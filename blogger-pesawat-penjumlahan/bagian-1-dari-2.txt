/* Q(terms, miss): miss = indeks bilangan yang disembunyikan (soal "angka hilang") */
function Q(terms,miss){
  var s=terms.reduce(function(a,b){return a+b;},0);
  if(miss==null) miss=-1;
  return {terms:terms,miss:miss,total:s,ans:miss>=0?terms[miss]:s};
}
function qText(q,reveal){
  var parts=q.terms.map(function(n,i){ return (!reveal&&i===q.miss)?'?':fmt(n,q.qAr); });
  var s=parts.join(' + ');
  if(q.miss>=0) s+=' = '+fmt(q.total,q.qAr);
  else if(reveal) s+=' = '+fmt(q.ans,q.qAr||q.ansAr);
  return s;
}
function sumTo(lo,hi){ var s=ri(lo,hi),a=ri(1,s-1); return Q([a,s-a]); }

}
function makeOpts(q){
  var a=q.ans, pref;
  if(q.tens) pref=[a+10,a-10,a+20,a-20];
  else if(q.carry) pref=[a-10,a+1,a-1,a+10];
  else if(a>=20) pref=[a+1,a-1,a+10,a-10,a+2];
  else pref=[a+1,a-1,a+2,a-2,a+3];
  pref=pref.filter(function(x,i,arr){ return x>=1&&x!==a&&arr.indexOf(x)===i; }).slice(0,4);
  var opts=[a];
  while(opts.length<3&&pref.length){ opts.push(pref.splice(Math.floor(Math.random()*pref.length),1)[0]); }
  var guard=0;
  while(opts.length<3&&guard++<200){ var x=ri(1,a+5); if(opts.indexOf(x)<0) opts.push(x); }
  for(var i=opts.length-1;i>0;i--){ var j=Math.floor(Math.random()*(i+1)); var t=opts[i]; opts[i]=opts[j]; opts[j]=t; }
  return opts;
}
function makeGate(){
  var q, guard=0;
  do{ q=newQuestion(); }while(qText(q)===lastKey&&guard++<30);
  lastKey=qText(q);
  var opts=makeOpts(q);
  return {y:hudH-gateH-12,opts:opts,correct:opts.indexOf(q.ans),state:'coming',pick:-1,q:q};
}
function spawnGate(){
  var g=makeGate(); gates.push(g);
  var el=$('#pa-target'), txt=qText(g.q);
  el.textContent=txt;
  el.classList.toggle('pa-arnum',g.q.qAr);
  el.classList.toggle('pa-long',txt.length>5&&txt.length<=8);
  el.classList.toggle('pa-xlong',txt.length>8);
  $('#pa-targetLabel').textContent='LEVEL '+(lvl+1)+' · '+(g.q.miss>=0?'ANGKA BERAPA?':'BERAPA HASILNYA?');
  el.classList.remove('pa-pop'); void el.offsetWidth; el.classList.add('pa-pop');
}
function travelTime(){ return Math.max(LV.tt*0.75,LV.tt-correctN*0.07)/(boost?3.5:1); }
function curSpeed(){
  if(state!=='run') return H*0.22;
  return ((car.y-car.h*0.5)-(hudH-12))/travelTime();
}
function setCorrect(){
  var el=$('#pa-points'); el.textContent=correctN+'/'+LV.goal;
  el.classList.remove('pa-bump'); void el.offsetWidth; el.classList.add('pa-bump');
}

function resolve(g){
  g.state='done'; g.pick=car.lane; total++; boost=false;
  if(g.pick===g.correct){
    correctN++; streak++;
    setCorrect();
    burst(laneX(g.correct),g.y+gateH*0.5,46);
    sfxWhoosh();
    addFloat('Benar!',car.x,car.y-car.h*0.9,'#1a8f4c',1.1,'big');
    if(streak>1) addFloat('Kombo x'+streak,car.x,car.y-car.h*0.9+40,'#d4560f',1.0);
    if(streak%5===0&&correctN<LV.goal){ timeLeft+=3; addFloat('Hebat! +3 detik',W/2,H*0.45,'#d4560f',1.6); burst(W/2,H*0.45,60,1.4); sfxHorn(); }
    if(correctN>=LV.goal){ endGame(true); return; }
  }else{
    streak=0; wrongN++; timeLeft=Math.max(0,timeLeft-3); shake=14; flash=1; car.wob=1;
    sfxThud();
    addFloat('−3 detik',car.x,car.y-car.h*0.9,'#e2483d',1.1,'big');
    var eq=qText(g.q,true);
    addFloat(eq,W/2,g.y-10,'#1a8f4c',1.8,'ar');
    if(mistakes.indexOf(eq)<0) mistakes.push(eq);
  }
  spawnGate();
}

var last=performance.now();
function frame(now){
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  update(dt); render(); renderFx(dt);
  requestAnimationFrame(frame);
}
function update(dt){
  if(state==='pause') return;
  speed+=(curSpeed()-speed)*Math.min(1,dt*(boost?6:3));
  scroll+=speed*dt;
  decor.forEach(function(d){ d.y+=speed*dt*(d.type==='tree'?d.sp:1); if(d.y>H+60){ var nd=newDecor(d.side,d.y-(H+240)); for(var k in nd) d[k]=nd[k]; } });
  var tx=laneX(car.lane);
  car.tilt+=(((tx-car.x)/road.lane)*0.35-car.tilt)*Math.min(1,dt*10);
  car.x+=(tx-car.x)*Math.min(1,dt*14);
  car.wob=Math.max(0,car.wob-dt*2.2);
  shake=Math.max(0,shake-dt*40); flash=Math.max(0,flash-dt*2.5);
  if(state==='run'||state==='count'){ dustT+=dt; if(dustT>(boost?0.03:0.06)){ dustT=0; dust.push({x:car.x+(Math.random()<.5?-1:1)*car.w*0.75,y:car.y+car.h*0.05,r:3+Math.random()*3,a:.45}); } }
  dust.forEach(function(p){ p.y+=speed*dt*0.9; p.r+=dt*14; p.a-=dt*0.9; });
  dust=dust.filter(function(p){return p.a>0;});
  /* peluru: antre tembakan lalu terbang ke atas */
  shots.forEach(function(s){ s.d-=dt; });
  shots=shots.filter(function(s){
    if(s.d>0) return true;
    gunSide=-gunSide;
    var pw=car.w*1.9;
    bullets.push({x:car.x+gunSide*pw*0.17,y:car.y-car.h*0.35,vy:-Math.max(900,H*1.7),life:0});
    sfxPew(); return false;
  });
  bullets.forEach(function(b){
    b.y+=b.vy*dt; b.life+=dt;
    gates.forEach(function(g){
      if(b.dead||g.state!=='coming') return;
      var k=Math.floor((b.x-road.x)/road.lane);
      if(k<0||k>2) return;
      if(b.y<=g.y+gateH&&b.y>=g.y-20){
        b.dead=true;
        burst(b.x,g.y+gateH,5,0.25);
        g.hit=(g.hit||0)+1; g.hitK=k; g.hitT=0.15;
      }
    });
  });
  gates.forEach(function(g){ if(g.hitT) g.hitT=Math.max(0,g.hitT-dt); });
  bullets=bullets.filter(function(b){ return !b.dead&&b.y>-30; });
  floats.forEach(function(f){ f.t+=dt; f.y-=dt*50; });
  floats=floats.filter(function(f){return f.t<f.life;});
  if(state==='run'){
    for(var i=0;i<gates.length&&state==='run';i++){ var g=gates[i]; g.y+=speed*dt; if(g.state==='coming'&&g.y+gateH>=car.y-car.h*0.5) resolve(g); }
    gates=gates.filter(function(g){return g.y<H+20;});
    if(state==='run'){
      timeLeft-=dt; updateTimer(); engineUpdate();
      if(timeLeft<=0){ timeLeft=0; updateTimer(); endGame(false); }
    }
  }else if(state==='end'){

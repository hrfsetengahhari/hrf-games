function noise(type){
  if(bufs[type]) return bufs[type];
  var len=AC.sampleRate*2, b=AC.createBuffer(1,len,AC.sampleRate), d=b.getChannelData(0), l=0;
  for(var i=0;i<len;i++){ var wv=Math.random()*2-1; if(type==='brown'){ l=(l+0.02*wv)/1.02; d[i]=l*3.5; } else d[i]=wv; }

function noiseShot(type,filt,f0,f1,dur,peak,q){
  if(!AC||!soundOn) return;
  var t=AC.currentTime, s=AC.createBufferSource(); s.buffer=noise(type);
  var f=AC.createBiquadFilter(); f.type=filt; f.Q.value=q||1;
  f.frequency.setValueAtTime(f0,t); f.frequency.exponentialRampToValueAtTime(f1,t+dur);
  var g=AC.createGain(); g.gain.setValueAtTime(0.0001,t); g.gain.exponentialRampToValueAtTime(peak,t+dur*0.25); g.gain.exponentialRampToValueAtTime(0.0001,t+dur);
  s.connect(f); f.connect(g); g.connect(AC.destination); s.start(t); s.stop(t+dur+0.05);
}

  var o=AC.createOscillator(); o.type='sawtooth'; o.frequency.value=70;
  var of=AC.createBiquadFilter(); of.type='lowpass'; of.frequency.value=260; of.Q.value=2;

  o.connect(of); of.connect(og); og.connect(master);

function startLevel(i){
  goFull(); initAudio();
  lvl=i; LV=LEVELS[i]; var my=++runId;
  hideAll();
  correctN=0;wrongN=0;total=0;streak=0;timeLeft=LV.time;mistakes=[];gates=[];floats=[];bullets=[];shots=[];lastKey='';car.lane=1;boost=false;
  $('#pa-points').textContent='0/'+LV.goal; $('#pa-target').textContent='–';
  $('#pa-target').className=''; $('#pa-targetLabel').textContent='LEVEL '+(i+1);
  updateTimer();
  $('#pa-hud').classList.remove('pa-hide'); floatBtns(true);
  state='count';
  engineStart();
  var go=function(){
    if(my!==runId||state!=='count') return;
    var seq=['3','2','1','Ayo!'], k=0;
    $('#pa-countLvl').textContent='Level '+(i+1)+' · '+LV.name;
    show('#pa-countdown',true);
    var step=function(){
      if(my!==runId||state!=='count') return;
      var n=$('#pa-countNum'); n.textContent=seq[k]; n.classList.remove('pa-go'); void n.offsetWidth; n.classList.add('pa-go');
      if(k===seq.length-1){ sfxHorn(); sfxWhoosh(); }
      k++;
      if(k<seq.length) setTimeout(step,700);
      else setTimeout(function(){
        if(my!==runId||state!=='count') return;
        show('#pa-countdown',false); layout(); state='run'; spawnGate();
        if(firstHint){ addFloat('Ketuk jalur untuk pindah!',W/2,H*0.55,'#1d2b3a',2.2); firstHint=false; }
      },600);
    };
    step();
  };
  var fontsReady=document.fonts?Promise.race([document.fonts.load('700 40px "Noto Naskh Arabic"','٣'),new Promise(function(r){setTimeout(r,1500);})]):Promise.resolve();
  fontsReady.then(function(){ setTimeout(function(){ layout(); go(); },200); },function(){ go(); });
}
function endGame(win){
  state='end'; boost=false; engineStop(); floatBtns(false);
  var stars=0, newUnlock=false;
  if(win){
    stars=wrongN===0?3:wrongN<=2?2:1;
    newUnlock=lvl+1<LEVELS.length&&!prog[lvl];
    if(stars>prog[lvl]){ prog[lvl]=stars; saveProg(); }
    sfxWin();
    for(var k=0;k<4;k++) setTimeout(function(){ burst(W*(0.2+Math.random()*0.6),H*(0.3+Math.random()*0.3),70,1.3); },k*250);
    rain(160);
  }else{
    sfxWind();
  }
  var acc=total?Math.round(correctN/total*100):0;
  var last=lvl===LEVELS.length-1;
  setTimeout(function(){
    if(state!=='end') return;
    $('#pa-hud').classList.add('pa-hide');
    $('#pa-endTitle').textContent=win?(last?'Semua level selesai! 🏆':'Level '+(lvl+1)+' selesai! 🏁'):'Waktu habis! ⏰';
    $('#pa-stars').textContent=starStr(stars);
    var st='Benar '+correctN+' dari '+total+' ('+acc+'%)';
    if(win&&newUnlock) st+=' · 🔓 Level '+(lvl+2)+' terbuka!';
    if(!win) st+=' · Butuh '+LV.goal+' jawaban benar';
    $('#pa-endStats').textContent=st;
    var rv=$('#pa-review'); rv.innerHTML='';
    mistakes.slice(0,8).forEach(function(m){ var c=document.createElement('div'); c.className='pa-chip'; c.textContent=m; rv.appendChild(c); });
    $('#pa-reviewWrap').style.display=mistakes.length?'flex':'none';
    var again=$('#pa-againBtn');
    if(win&&!last){ again.textContent='LEVEL '+(lvl+2)+' ▶'; again.dataset.act='next'; }
    else if(win&&last){ again.textContent='PILIH LEVEL ▶'; again.dataset.act='menu'; }
    else { again.textContent='COBA LAGI ▶'; again.dataset.act='retry'; }
    $('#pa-retryBtn').style.display=win?'':'none';
    show('#pa-endScreen',true);
  },900);
}

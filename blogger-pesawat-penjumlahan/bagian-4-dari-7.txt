    addFloat('Benar!',car.x,car.y-car.h*0.9,'#1a8f4c',1.1,'big');
    if(streak>1) addFloat('Kombo x'+streak,car.x,car.y-car.h*0.9+40,'#d4560f',1.0);
    if(streak%5===0&&correctN<LV.goal){ timeLeft+=3; addFloat('Hebat! +3 detik',W/2,H*0.45,'#d4560f',1.6); burst(W/2,H*0.45,60,1.4); sfxHorn(); }
    if(correctN>=LV.goal){ endGame(true); return; }
  }else{
    streak=0; wrongN++; timeLeft=Math.max(0,timeLeft-3); shake=14; flash=1; car.wob=1;
    sfxThud();
    addFloat('−3 detik',car.x,car.y-car.h*0.9,'#e2483d',1.1,'big');
    var eq=qText(g.q,true);
    addFloat(eq,W/2,g.y-10,'#1a8f4c',1.8,'ar');
    if(mistakes.indexOf(eq)<0) mistakes.push(eq);
  }
  spawnGate();
}

var last=performance.now();
function frame(now){
  var dt=Math.min(0.05,(now-last)/1000); last=now;
  update(dt); render(); renderFx(dt);
  requestAnimationFrame(frame);
}
function update(dt){
  if(state==='pause') return;
  speed+=(curSpeed()-speed)*Math.min(1,dt*(boost?6:3));
  scroll+=speed*dt;
  decor.forEach(function(d){ d.y+=speed*dt*(d.type==='tree'?d.sp:1); if(d.y>H+60){ var nd=newDecor(d.side,d.y-(H+240)); for(var k in nd) d[k]=nd[k]; } });
  var tx=laneX(car.lane);
  car.tilt+=(((tx-car.x)/road.lane)*0.35-car.tilt)*Math.min(1,dt*10);
  car.x+=(tx-car.x)*Math.min(1,dt*14);
  car.wob=Math.max(0,car.wob-dt*2.2);
  shake=Math.max(0,shake-dt*40); flash=Math.max(0,flash-dt*2.5);
  if(state==='run'||state==='count'){ dustT+=dt; if(dustT>(boost?0.03:0.06)){ dustT=0; dust.push({x:car.x+(Math.random()<.5?-1:1)*car.w*0.75,y:car.y+car.h*0.05,r:3+Math.random()*3,a:.45}); } }
  dust.forEach(function(p){ p.y+=speed*dt*0.9; p.r+=dt*14; p.a-=dt*0.9; });
  dust=dust.filter(function(p){return p.a>0;});
  /* peluru: antre tembakan lalu terbang ke atas */
  shots.forEach(function(s){ s.d-=dt; });
  shots=shots.filter(function(s){
    if(s.d>0) return true;
    gunSide=-gunSide;
    var pw=car.w*1.9;
    bullets.push({x:car.x+gunSide*pw*0.17,y:car.y-car.h*0.35,vy:-Math.max(900,H*1.7),life:0});
    sfxPew(); return false;
  });
  bullets.forEach(function(b){
    b.y+=b.vy*dt; b.life+=dt;
    gates.forEach(function(g){
      if(b.dead||g.state!=='coming') return;
      var k=Math.floor((b.x-road.x)/road.lane);
      if(k<0||k>2) return;
      if(b.y<=g.y+gateH&&b.y>=g.y-20){
        b.dead=true;
        burst(b.x,g.y+gateH,5,0.25);
        g.hit=(g.hit||0)+1; g.hitK=k; g.hitT=0.15;
      }
    });
  });
  gates.forEach(function(g){ if(g.hitT) g.hitT=Math.max(0,g.hitT-dt); });
  bullets=bullets.filter(function(b){ return !b.dead&&b.y>-30; });
  floats.forEach(function(f){ f.t+=dt; f.y-=dt*50; });
  floats=floats.filter(function(f){return f.t<f.life;});
  if(state==='run'){
    for(var i=0;i<gates.length&&state==='run';i++){ var g=gates[i]; g.y+=speed*dt; if(g.state==='coming'&&g.y+gateH>=car.y-car.h*0.5) resolve(g); }
    gates=gates.filter(function(g){return g.y<H+20;});
    if(state==='run'){
      timeLeft-=dt; updateTimer(); engineUpdate();
      if(timeLeft<=0){ timeLeft=0; updateTimer(); endGame(false); }
    }
  }else if(state==='end'){
    gates.forEach(function(g){ g.y+=speed*dt; });
  }
}

  var s=Math.ceil(timeLeft);
  $('#pa-time').textContent=s; $('#pa-timer').classList.toggle('pa-low',s<=10);
  $('#pa-timefill').style.width=Math.min(100,timeLeft/LV.time*100)+'%';
}

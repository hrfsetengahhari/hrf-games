function engineStart(){
  if(!AC||!soundOn||eng) return;
  var t=AC.currentTime;
  var master=AC.createGain(); master.gain.setValueAtTime(0.0001,t); master.gain.exponentialRampToValueAtTime(1,t+0.5); master.connect(AC.destination);
  // gemuruh mesin
  var ns=AC.createBufferSource(); ns.buffer=noise('brown'); ns.loop=true;
  var nf=AC.createBiquadFilter(); nf.type='lowpass'; nf.frequency.value=320;
  var ng=AC.createGain(); ng.gain.value=0.16;
  ns.connect(nf); nf.connect(ng); ng.connect(master);
  // dengung putaran mesin
  var o=AC.createOscillator(); o.type='sawtooth'; o.frequency.value=70;
  var of=AC.createBiquadFilter(); of.type='lowpass'; of.frequency.value=260; of.Q.value=2;

  var lfo=AC.createOscillator(); lfo.frequency.value=22; var lg=AC.createGain(); lg.gain.value=0.04;
  lfo.connect(lg); lg.connect(og.gain);
  o.connect(of); of.connect(og); og.connect(master);
  ns.start(); o.start(); lfo.start();
  eng={master:master,ns:ns,o:o,lfo:lfo,of:of,nf:nf};
}

var nativeFull=false, prevOverflow='';
function goFull(){
  if(!app.classList.contains('pa-full')){
    prevOverflow=document.documentElement.style.overflow;
    document.documentElement.style.overflow='hidden';
    app.classList.add('pa-full');
  }
  var isFull=document.fullscreenElement||document.webkitFullscreenElement;
  if(!isFull){
    var rf=app.requestFullscreen||app.webkitRequestFullscreen;
    if(rf){ try{ var p=rf.call(app,{navigationUI:'hide'}); nativeFull=true; if(p&&p.catch) p.catch(function(){ nativeFull=false; }); }catch(e){ nativeFull=false; } }
  }
  try{ screen.orientation&&screen.orientation.lock&&screen.orientation.lock('portrait').catch(function(){}); }catch(e){}
  setTimeout(resize,150);
}
function exitFull(){
  app.classList.remove('pa-full');
  document.documentElement.style.overflow=prevOverflow;
  if(document.fullscreenElement||document.webkitFullscreenElement){
    var ef=document.exitFullscreen||document.webkitExitFullscreen; try{ var p=ef.call(document); if(p&&p.catch) p.catch(function(){}); }catch(e){}
  }
  nativeFull=false;
  setTimeout(resize,150);
}
function onFsChange(){
  if(!(document.fullscreenElement||document.webkitFullscreenElement)&&nativeFull){
    nativeFull=false; if(state==='run') pause(); exitFull();
  }
  setTimeout(resize,150);
}

function starStr(n){ return new Array(n+1).join('⭐')+new Array(4-n).join('☆'); }
function renderLevels(){
  var box=$('#pa-levels'); box.innerHTML='';
  var nextIdx=-1;
  for(var i=0;i<LEVELS.length;i++) if(unlocked(i)&&!prog[i]){ nextIdx=i; break; }
  LEVELS.forEach(function(L,i){
    var b=document.createElement('button'), open=unlocked(i);
    b.className='pa-lvl'+(open?'':' pa-locked')+(i===nextIdx?' pa-next':'');
    b.setAttribute('aria-label','Level '+(i+1)+': '+L.name+(open?'':' (terkunci)'));
    b.innerHTML=open
      ? '<b>'+(i+1)+'</b><span>'+L.name+'</span><i>'+starStr(prog[i])+'</i>'
      : '<b>🔒</b><span>Level '+(i+1)+'</span><i>&nbsp;</i>';
    b.addEventListener('click',function(){
      if(!unlocked(i)){
        initAudio(); sfxThud();
